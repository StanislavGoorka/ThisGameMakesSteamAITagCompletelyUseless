cmake_minimum_required(VERSION 3.26)
project(ClosedCircuit LANGUAGES CXX)

set(CMAKE_CXX_EXTENSIONS OFF)

add_library(game_core INTERFACE)
target_compile_features(game_core INTERFACE cxx_std_20)
target_include_directories(game_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")

option(CLOSED_CIRCUIT_PUBLIC_BUILD "Require approved public content" OFF)
target_compile_definitions(game_core INTERFACE CLOSED_CIRCUIT_PUBLIC_BUILD=$<BOOL:${CLOSED_CIRCUIT_PUBLIC_BUILD}>)
if(WIN32)
    if(CLOSED_CIRCUIT_PUBLIC_BUILD)
        set(content_gate "Validate-PublicContent.ps1")
    else()
        set(content_gate "Validate-AssetPackage.ps1")
    endif()
    add_custom_target(validate_content ALL
        COMMAND powershell -NoProfile -ExecutionPolicy Bypass
            -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/${content_gate}"
            -Root "${CMAKE_CURRENT_SOURCE_DIR}"
        VERBATIM)
    add_dependencies(game_core validate_content)
endif()

    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "content/texts.json")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/content/texts.json" text_registry)
    string(JSON text_count LENGTH "${text_registry}" texts)
    set(developer_records "")
    if(text_count GREATER 0)
        math(EXPR text_last "${text_count} - 1")
        foreach(index RANGE 0 ${text_last})
            string(JSON text_id GET "${text_registry}" texts ${index} id)
            string(JSON text_status GET "${text_registry}" texts ${index} status)
            if(NOT text_status STREQUAL "ai_placeholder")
                message(FATAL_ERROR "Developer catalog requires marked records")
            endif()
            string(JSON text_value GET "${text_registry}" texts ${index} value)
            string(FIND "${text_value}" ")CC_TEXT\"" delimiter_position)
            if(NOT delimiter_position EQUAL -1)
                message(FATAL_ERROR "Unsupported raw text delimiter")
            endif()
            string(APPEND developer_records "    TextRecord{\"${text_id}\", R\"CC_TEXT(${text_value})CC_TEXT\", TextStatus::ai_placeholder, std::nullopt},\n")
        endforeach()
    endif()
    configure_file(src/platform/windows/developer_text.hpp.in generated/developer_text.hpp @ONLY)
